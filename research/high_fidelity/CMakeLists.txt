cmake_minimum_required(VERSION 3.18)
project(MovingTankPrototype LANGUAGES CXX)
find_package(Chrono CONFIG REQUIRED COMPONENTS FSI_SPH)
add_executable(moving_tank moving_tank.cpp)
add_executable(test_moving_motion test_moving_motion.cpp)
foreach(target moving_tank test_moving_motion)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_link_libraries(${target} PRIVATE ${CHRONO_TARGETS})
endforeach()
enable_testing()
add_test(NAME analytic_and_no_fluid_motion COMMAND test_moving_motion)
